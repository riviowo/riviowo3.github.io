import { POLYGON_CHAIN_ID, POLYGON_RPC_CANDIDATES } from "../config.js";

let activeRpc = null;

async function rpcFetch(url, method, params = [], timeoutMs = 6500) {
  const ctrl = new AbortController();
  const t = setTimeout(() => ctrl.abort(), timeoutMs);
  try {
    const res = await fetch(url, {
      method: "POST",
      signal: ctrl.signal,
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ jsonrpc: "2.0", id: 1, method, params }),
    });
    if (!res.ok) throw new Error(`RPC HTTP ${res.status}`);
    const json = await res.json();
    if (json.error) throw new Error(json.error.message || "RPC error");
    return json.result;
  } finally {
    clearTimeout(t);
  }
}

async function isHealthy(url) {
  const chainId = await rpcFetch(url, "eth_chainId", []);
  if (String(chainId).toLowerCase() !== POLYGON_CHAIN_ID) throw new Error("Wrong chainId");
  await rpcFetch(url, "eth_blockNumber", []);
  return true;
}

export async function pickHealthyPolygonRpc() {
  for (const url of POLYGON_RPC_CANDIDATES) {
    try {
      await isHealthy(url);
      activeRpc = url;
      return url;
    } catch {}
  }
  throw new Error("No healthy Polygon RPC found.");
}

export async function rpcRequestPolygon(method, params = []) {
  if (!activeRpc) await pickHealthyPolygonRpc();
  try {
    return await rpcFetch(activeRpc, method, params, 9000);
  } catch {
    activeRpc = null;
    await pickHealthyPolygonRpc();
    return await rpcFetch(activeRpc, method, params, 9000);
  }
}
